% Generic proof. Its application to all 23 types is conditional until both
% hard-component Hamilton paths have been independently verified.

\begin{lemma}[Protected products with a traceable second factor]
Let $B$ have a Hamilton path of order at least two. Suppose that $A\square P_2$
and $A\square P_3$ have Hamilton cycles retaining every edge of $K\subseteq E(A)$
and one fixed edge $e\in E(A)\setminus K$ in every physical path row.
Then $A\square B$ has a Hamilton cycle retaining every edge of $K$ in every
physical $B$ row.
\end{lemma}
\begin{proof}
Partition a Hamilton path of $B$ into consecutive paths of order two,
using one path of order three if its order is odd. On each part use its
specified product cycle. Join consecutive parts by deleting the two copies
of $e$ in their endpoint rows and adding the two cross edges supplied by
the intervening Hamilton-path edge of $B$.
Each physical row participates in at most one such join: the first and
last rows of every part are distinct. Thus every required copy of $e$
is still present when used. Processing the parts in order merges their
disjoint cycles into one. No edge of $K$ is deleted.
\end{proof}
